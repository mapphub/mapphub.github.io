Обзор Манифест MiniApp API Публикация

mapp.manifest.json

Каноническое имя. Оболочка также принимает mappmanifest.json.

Пример

{
  "id": "chat",
  "name": "Чат",
  "version": "1.2.0",
  "description": "LAN-чат",
  "author": "mapphub",
  "icon": "icon.svg",
  "routes": { "root": "public", "app": "public", "client": "public" },
  "server": { "host": true, "client": true, "network": true },
  "permissions": ["net", "share", "db", "notify"],
  "tags": ["social"],
  "launcher": false
}
